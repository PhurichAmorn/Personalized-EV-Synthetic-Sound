cmake_minimum_required(VERSION 3.16)
project(evsound LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(evsound STATIC config.cpp dsp.cpp engine.cpp order.cpp random.cpp output.cpp)
target_compile_features(evsound PUBLIC cxx_std_17)
target_include_directories(evsound PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(evsound PUBLIC Threads::Threads ${CMAKE_DL_LIBS})

add_executable(evsound_render render_main.cpp)
target_link_libraries(evsound_render PRIVATE evsound)

include(CTest)
if(BUILD_TESTING)
    add_library(evsound_fake_portaudio SHARED ../tests/fake_portaudio.cpp)
    target_compile_features(evsound_fake_portaudio PRIVATE cxx_std_17)
    set_target_properties(evsound_fake_portaudio PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
    add_executable(evsound_output_test ../tests/native_output_test.cpp)
    target_link_libraries(evsound_output_test PRIVATE evsound)
    # Lifecycle checks use assert and must also run in Release builds.
    if(MSVC)
        target_compile_options(evsound_output_test PRIVATE /UNDEBUG)
        target_compile_options(evsound_fake_portaudio PRIVATE /UNDEBUG)
    else()
        target_compile_options(evsound_output_test PRIVATE -UNDEBUG)
        target_compile_options(evsound_fake_portaudio PRIVATE -UNDEBUG)
    endif()
    add_executable(evsound_engine_test ../tests/native_engine_test.cpp)
    target_link_libraries(evsound_engine_test PRIVATE evsound)
    foreach(test_target evsound_output_test evsound_engine_test)
        if(MSVC)
            target_compile_options(${test_target} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${test_target} PRIVATE -UNDEBUG)
        endif()
    endforeach()
    add_test(NAME native_engine
             COMMAND evsound_engine_test $<TARGET_FILE:evsound_fake_portaudio>)
    add_test(NAME native_output_lifecycle
             COMMAND evsound_output_test $<TARGET_FILE:evsound_fake_portaudio>)
endif()
